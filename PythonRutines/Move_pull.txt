
def move(poss, variable_dict):
    # One pull step: the gap grows by dz (each electrode moves dz/2 if both)
    global Nbtm, Nmol, Ntop, dz, both
    i = 0
    used = []
    step = dz/2 if both else dz
    while i < Nbtm:
        if both:
            try:
                poss[i][-1] = float(poss[i][-1])-step
            except:
                if not poss[i][-1] in used:
                    used.append(poss[i][-1])
                    variable_dict[poss[i][-1]] -= step
        i+=1

    i += Nmol

    while i < Nbtm+Nmol+Ntop:
        try:
            poss[i][-1] = float(poss[i][-1])+step
        except:
            if not poss[i][-1] in used:
                used.append(poss[i][-1])
                variable_dict[poss[i][-1]] += step
        i+=1
    return poss, variable_dict
