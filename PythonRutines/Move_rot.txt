

import numpy as np
dtheta *= 2*np.pi/360
# Row-vector convention, as in ANT.UI's viewer: r' = r @ rotationZi turns counter-clockwise by dtheta
rotationZi = np.array([[np.cos(dtheta), np.sin(dtheta), 0],
                        [-np.sin(dtheta), np.cos(dtheta), 0],
                        [0, 0, 1]])

def move(poss, variable_dict):
    # Rotate each selected component about the z axis through its own centre
    global Nbtm, Nmol, Ntop, rotationZi, Flag_bottom, Flag_mol, Flag_top
    groups = [(0, Nbtm, Flag_bottom), (Nbtm, Nbtm+Nmol, Flag_mol), (Nbtm+Nmol, Nbtm+Nmol+Ntop, Flag_top)]

    def value(token):
        try:
            return float(token)
        except (TypeError, ValueError):
            return variable_dict[token]

    for start, end, flag in groups:
        if not flag:
            continue
        xyz = np.array([[value(t) for t in poss[i][-3:]] for i in range(start, end)])
        centre = xyz.mean(axis = 0)
        centre[2] = 0
        new = (xyz - centre) @ rotationZi + centre
        used = []
        for k, i in enumerate(range(start, end)):
            for a in range(3):
                token = poss[i][len(poss[i])-3+a]
                try:
                    float(token)
                    poss[i][len(poss[i])-3+a] = new[k][a]
                except (TypeError, ValueError):
                    if not token in used:
                        used.append(token)
                        variable_dict[token] = new[k][a]
    return poss, variable_dict
